QUICK DEPLOY (automated — run once on a fresh Ubuntu server)
─────────────────────────────────────────────────────────────
  If you already have SSH access to the server:

    1. Copy the deploy script to the server:
         scp deploy/deploy-server.sh ubuntu@130.61.83.110:~/

    2. SSH into the server:
         ssh ubuntu@130.61.83.110

    3. Run the script:
         chmod +x deploy-server.sh
         bash deploy-server.sh

  That's it — the script installs everything and deploys both
  frontend and backend automatically.

  For Windows machines using PuTTY/CyberArk, use the manual
  steps in PART 1–4 below instead.

================================================================================
PART 1 — ONE-TIME SERVER SETUP
================================================================================

-- 1.1  Connect to the server ------------------------------------------------

  Via SSH (Linux/Mac/Windows Terminal):
    ssh ubuntu@130.61.83.110

  Via PuTTY (Windows):
    Host: 130.61.83.110   Port: 22   Connection type: SSH
    Load your .ppk key under Connection > SSH > Auth > Credentials

  Via CyberArk/Guacamole:
    Use the web terminal as you normally do.


-- 1.2  Update system packages -----------------------------------------------

  sudo apt-get update -y && sudo apt-get upgrade -y


-- 1.3  Install Node.js 20 ---------------------------------------------------

  curl -fsSL https://deb.nodesource.com/setup_20.x | sudo -E bash -
  sudo apt-get install -y nodejs
  node -v    # v20.x.x
  npm -v


-- 1.4  Install Git, Nginx, PM2 ---------------------------------------------

  sudo apt-get install -y nginx git
  sudo npm install -g pm2
  pm2 -v
  nginx -v


-- 1.5  Install MySQL 8 ------------------------------------------------------

  sudo apt-get install -y mysql-server
  sudo systemctl enable mysql
  sudo systemctl start mysql

  # Secure MySQL (set root password, remove test DBs):
  sudo mysql_secure_installation

  # If mysql_secure_installation has issues, set root password manually:
  sudo mysql -e "ALTER USER 'root'@'localhost' IDENTIFIED WITH mysql_native_password BY 'YourPassword'; FLUSH PRIVILEGES;"

  # Verify:
  mysql -u root -p
  EXIT;


-- 1.6  Create app directories -----------------------------------------------

  sudo mkdir -p /var/www/car-workshop/backend
  sudo mkdir -p /var/www/car-workshop/frontend
  sudo chown -R ubuntu:ubuntu /var/www/car-workshop


-- 1.7  Configure Ubuntu firewall (ufw) --------------------------------------

  sudo ufw allow OpenSSH
  sudo ufw allow 'Nginx HTTP'
  sudo ufw allow 'Nginx HTTPS'
  sudo ufw enable
  sudo ufw status

PART 2 — DEPLOY BACKEND

  cd /var/www/car-workshop
  git clone https://github.com/osamahkenawy/car-workshop-backend.git backend

-- 2.2  Install dependencies -------------------------------------------------

  npm install

-- 2.4  Create MySQL database and run migrations -----------------------------

  # Create database:
  mysql -u root -p -e "CREATE DATABASE IF NOT EXISTS car_workshop CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;"

  # Run main schema migration:
  mysql -u root -p car_workshop < /var/www/car-workshop/backend/src/migrations/car_workshop.sql

  # Run post-migration patches:
  mysql -u root -p car_workshop < /var/www/car-workshop/backend/src/migrations/post/00_schema_patches.sql
  mysql -u root -p car_workshop < /var/www/car-workshop/backend/src/migrations/post/01_seed_countries.sql

  # Verify tables were created:
  mysql -u root -p -e "SHOW TABLES;" car_workshop


-- 2.5  Start backend with PM2 -----------------------------------------------

  cd /var/www/car-workshop/backend

  pm2 start src/server.js \
    --name car-workshop-backend \
    --max-memory-restart 512M

  pm2 status
  pm2 logs car-workshop-backend --lines 50    # check for startup errors

  # Save and enable PM2 on reboot:
  pm2 save
  pm2 startup systemd -u ubuntu --hp /home/ubuntu
  # Run the printed "sudo env PATH=..." command


-- 2.6  Verify backend is running -------------------------------------------

  curl http://127.0.0.1:4000/health
  # Expected: {"success":true,"service":"car-workshop-backend","status":"ok"}

  curl http://127.0.0.1:4000/api/docs
  # Expected: HTML page (Swagger UI)

================================================================================
PART 4 — CONFIGURE NGINX
================================================================================

-- 4.1  Create Nginx site config --------------------------------------------

  sudo tee /etc/nginx/sites-available/car-workshop > /dev/null << 'EOF'
server {
    listen 80;
    server_name workshop.pioneeruae.com 130.61.83.110;

    # ── Frontend SPA ───────────────────────────────────────────
    root /var/www/car-workshop/frontend;
    index index.html;

    location / {
        try_files $uri $uri/ /index.html;
    }

    # ── Backend REST API + Swagger UI ──────────────────────────
    location /api/ {
        proxy_pass         http://127.0.0.1:4000/api/;
        proxy_http_version 1.1;
        proxy_set_header   Upgrade           $http_upgrade;
        proxy_set_header   Connection        'upgrade';
        proxy_set_header   Host              $host;
        proxy_set_header   X-Real-IP         $remote_addr;
        proxy_set_header   X-Forwarded-For   $proxy_add_x_forwarded_for;
        proxy_set_header   X-Forwarded-Proto $scheme;
        proxy_read_timeout 120s;
    }

    # ── Socket.IO ─────────────────────────────────────────────
    location /socket.io/ {
        proxy_pass         http://127.0.0.1:4000/socket.io/;
        proxy_http_version 1.1;
        proxy_set_header   Upgrade    $http_upgrade;
        proxy_set_header   Connection "upgrade";
        proxy_set_header   Host       $host;
        proxy_cache_bypass $http_upgrade;
    }

    # ── Uploaded files ────────────────────────────────────────
    location /uploads/ {
        proxy_pass       http://127.0.0.1:4000/uploads/;
        proxy_set_header Host $host;
    }

    client_max_body_size 50M;

    gzip on;
    gzip_types text/plain text/css application/json application/javascript
               text/xml application/xml image/svg+xml;
    gzip_min_length 1000;
}
EOF


-- 4.2  Enable site and reload Nginx -----------------------------------------

  sudo ln -sf /etc/nginx/sites-available/car-workshop \
              /etc/nginx/sites-enabled/car-workshop

  # Remove default Nginx page (optional, only if no other apps on server):
  # sudo rm -f /etc/nginx/sites-enabled/default

  sudo nginx -t                          # test config — must say "ok"
  sudo systemctl reload nginx

PART 7 — SWAGGER / API DOCS
================================================================================

  The Swagger UI is served directly by the backend at:

    http://workshop.pioneeruae.com/api/docs   ← via Nginx proxy
    http://127.0.0.1:4000/api/docs            ← direct on server

  What's documented:
    • 393 API paths across 35 tags
    • Auth (JWT Bearer) and External API (X-API-Key) security schemes
    • All route groups: Auth, Work Orders, Mechanics, Customers,
      Vehicles, Invoices, Wallet, Reports, Super Admin, and more

  To use protected endpoints in the Swagger UI:
    1. Call POST /api/auth/login to get a JWT token
    2. Click "Authorize" (lock icon) in the top right
    3. Enter:  Bearer <your-token>
    4. All authenticated requests will include the token

  The spec file is at:
    /var/www/car-workshop/backend/src/swagger.yaml

PART 8 — TROUBLESHOOTING
================================================================================

PROBLEM: npm install fails with EACCES / permission error
  sudo chown -R ubuntu:ubuntu /var/www/car-workshop
  cd /var/www/car-workshop/backend && npm install

PROBLEM: Backend crash on startup — "Cannot find module 'swagger-ui-express'"
  cd /var/www/car-workshop/backend
  npm install     # NOT --omit=dev, swagger packages are in dependencies

PROBLEM: curl http://127.0.0.1:4000/health → Connection refused
  pm2 logs car-workshop-backend --lines 100
  # Check for DB connection error, missing .env, or port conflict

PROBLEM: Nginx 502 Bad Gateway
  pm2 status
  pm2 restart car-workshop-backend
  sudo systemctl reload nginx

PROBLEM: Nginx 404 on /api/docs
  # Make sure Nginx is proxying /api/ correctly:
  sudo nginx -t
  curl http://127.0.0.1:4000/api/docs    # test backend directly

PROBLEM: MySQL "ERROR 1045: Access denied for user 'root'"
  sudo mysql -e "ALTER USER 'root'@'localhost' IDENTIFIED WITH mysql_native_password BY 'NewPassword'; FLUSH PRIVILEGES;"
  # Then update DB_PASSWORD in .env and: pm2 restart car-workshop-backend

PROBLEM: Frontend builds but shows blank page
  # Check browser console for errors
  # Ensure VITE_API_URL was set before running npm run build
  cat /tmp/frontend-src/.env.production

PROBLEM: PM2 not running after server reboot
  pm2 startup systemd -u ubuntu --hp /home/ubuntu
  # Run the printed sudo command, then:
  pm2 save

PROBLEM: Port 80 not reachable from outside
  sudo ufw status
  sudo ufw allow 'Nginx HTTP'
  sudo ufw reload
